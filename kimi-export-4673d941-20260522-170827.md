   237	function SimpleSlider({value, min, max, onChange}: {value: number; min: number; max: number; onChange: (v: number) => void}) {
   238	  const pct = (value - min) / (max - min);
   239	  const [trackWidth, setTrackWidth] = React.useState(0);
   240	  return (
   241	    <TouchableOpacity
   242	      style={styles.sliderTrack}
   243	      activeOpacity={1}
   244	      onLayout={(e) => setTrackWidth(e.nativeEvent.layout.width)}
   245	      onPress={(e) => {
   246	        const locationX = e.nativeEvent.locationX;
   247	        const newPct = Math.max(0, Math.min(1, locationX / trackWidth));
   248	        onChange(min + newPct * (max - min));
   249	      }}>
   250	      <View style={[styles.sliderFill, {width: `${pct * 100}%`}]} />
   251	      <View style={[styles.sliderThumb, {left: `${pct * 100}%`, marginLeft: -10}]} />
   252	    </TouchableOpacity>
   253	  );
   254	}
